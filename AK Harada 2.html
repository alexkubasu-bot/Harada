                const taskId = `${pillarKey}-${idx}`;

                const isDaily = task.daily;

                const isCompleted = isDaily ? isDailyDoneToday(taskId) : (taskStatus[taskId] || false);


                // For daily tasks, show how many days completed total

                let streakBadge = '';

                if (isDaily) {

                    const daysCount = dailyStatus[taskId] ? Object.values(dailyStatus[taskId]).filter(Boolean).length : 0;

                    streakBadge = `<span class="text-xs px-1.5 py-0.5 rounded-full bg-blue-100 dark:bg-blue-900/40 text-blue-600 dark:text-blue-300 font-semibold whitespace-nowrap">${daysCount}d</span>`;

                }

                

                return `

                    <div class="task-row flex items-center gap-3 p-3 rounded-lg ${isCompleted ? 'bg-emerald-50 dark:bg-emerald-900/20' : 'bg-white dark:bg-slate-800/50'} border border-slate-100 dark:border-slate-700 cursor-pointer group" onclick="toggleTask('${taskId}', ${isDaily})">

                        <input type="checkbox" class="checkbox-harada" ${isCompleted ? 'checked' : ''} onclick="event.stopPropagation(); toggleTask('${taskId}', ${isDaily})">

                        <span class="text-sm ${isCompleted ? 'line-through text-slate-400' : 'text-slate-700 dark:text-slate-300'} flex-1">${task.text}</span>

                        ${isDaily ? `<span class="text-xs px-2 py-0.5 rounded-full bg-indigo-100 dark:bg-indigo-900/40 text-indigo-600 dark:text-indigo-300 font-semibold whitespace-nowrap">Daily</span>` : `<span class="text-xs px-2 py-0.5 rounded-full bg-slate-100 dark:bg-slate-700 text-slate-500 font-medium whitespace-nowrap">Once</span>`}

                        ${streakBadge}

                        ${isCompleted ? '<i data-lucide="check-circle" class="w-5 h-5 text-emerald-500 shrink-0"></i>' : ''}

                    </div>

                `;

            }).join('');

            

            lucide.createIcons();

            updatePillarPercentage(pillarKey, tasks.length);

        }


        function toggleTask(taskId, isDaily) {

            if (isDaily) {

                // Toggle today's entry for daily tasks

                if (!dailyStatus[taskId]) dailyStatus[taskId] = {};

                const today = todayKey();

                dailyStatus[taskId][today] = !dailyStatus[taskId][today];

                localStorage.setItem('haradaDaily', JSON.stringify(dailyStatus));

            } else {

                taskStatus[taskId] = !taskStatus[taskId];

                localStorage.setItem('haradaTasks', JSON.stringify(taskStatus));

            }

            

            const pillarKey = taskId.split('-')[0];

            renderPillar(pillarKey, haradaData[pillarKey]);

            renderVisualGrid();

            updateStats();

            

            // Celebrate if all 8 tasks in the pillar are done today/overall

            const pillarDone = haradaData[pillarKey].filter((task, idx) => {

                const tid = `${pillarKey}-${idx}`;

                return task.daily ? isDailyDoneToday(tid) : (taskStatus[tid] || false);

            }).length;

            if (pillarDone === 8) {

                celebratePillarCompletion(pillarKey);

            }

        }


        function updatePillarPercentage(pillarKey, total) {

            const completed = haradaData[pillarKey].filter((task, idx) => {

                const tid = `${pillarKey}-${idx}`;

                return task.daily ? isDailyDoneToday(tid) : (taskStatus[tid] || false);

            }).length;

            const percentage = Math.round((completed / total) * 100);

            

            const percentEl = document.getElementById(pillarKey + 'Percent');

            const scoreEl = document.getElementById(pillarKey + 'Score');

            const countEl = document.getElementById(pillarKey + 'Count');

            

            if (percentEl) percentEl.textContent = percentage + '%';

            if (scoreEl) scoreEl.textContent = percentage + '%';

            if (countEl) countEl.textContent = `${completed}/${total}`;

        }


        function renderSixMonthCalendar() {

            const today = new Date();

                const d = new Date(today.getFullYear(), today.getMonth() + i, 1);

                months.push({

                    name: d.toLocaleString('default', { month: 'long' }),

                    shortName: d.toLocaleString('default', { month: 'short' }),

                    year: d.getFullYear(),

                    days: new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate(),

                    key: `${d.getFullYear()}-${d.getMonth()}`

                });

            }

                const monthData = calendarData[month.key] || {};

                const completedDays = Object.values(monthData).filter(v => v).length;

                const percentage = Math.round((completedDays / month.days) * 100);

                const isCurrentMonth = idx === 0;

                

                return `

                            <span>S</span><span>M</span><span>T</span><span>W</span><span>T</span><span>F</span><span>S</span>

                        <div class="grid grid-cols-7 gap-1">

                            ${Array.from({length: month.days}, (_, i) => {

                                const day = i + 1;

                                const isCompleted = monthData[day];

                                const isToday = isCurrentMonth && day === today.getDate();

                                

                                return `

                                    <button class="calendar-day aspect-square rounded-lg flex items-center justify-center text-xs font-medium

                                        ${isCompleted ? 'bg-emerald-500 text-white shadow-md' : 'bg-slate-100 dark:bg-slate-700 text-slate-600 dark:text-slate-400 hover:bg-slate-200 dark:hover:bg-slate-600'}

                                        ${isToday ? 'ring-2 ring-blue-500 ring-offset-2 dark:ring-offset-slate-800 bg-blue-50 dark:bg-blue-900/30 text-blue-600' : ''}"

                        </div>

            updateStreakStats();

        }


        function toggleCalendarDay(monthKey, day) {

            if (!calendarData[monthKey]) calendarData[monthKey] = {};

            calendarData[monthKey][day] = !calendarData[monthKey][day];

            localStorage.setItem('haradaCalendar', JSON.stringify(calendarData));

            renderSixMonthCalendar();

            updateStats();

            

            const streak = calculateCurrentStreak();

            if (calendarData[monthKey][day] && (streak === 7 || streak === 30 || streak === 100)) {

                celebrateStreak(streak);

            }

        }


        function calculateCurrentStreak() {

            let streak = 0;

            const today = new Date();

            const checkDate = new Date(today);

            

            while (true) {

                const monthKey = `${checkDate.getFullYear()}-${checkDate.getMonth()}`;

                const day = checkDate.getDate();

                const monthData = calendarData[monthKey] || {};

                

                if (monthData[day]) {

                    streak++;

                    checkDate.setDate(checkDate.getDate() - 1);

                } else {

                    break;

                }

            }

            

            return streak;

        }


        function updateStreakStats() {

            const streak = calculateCurrentStreak();

            let total = 0;

            Object.values(calendarData).forEach(month => {

                total += Object.values(month).filter(v => v).length;

            });

            

            document.getElementById('currentStreak').textContent = streak;

            document.getElementById('totalMarked').textContent = total;

        }


        function renderVisualGrid() {

            const container = document.getElementById('visualGrid');

            const pillars = ['body', 'control', 'technique', 'digital', 'mind', 'spirit', 'karma', 'team'];

            

            container.innerHTML = pillars.map(pillar => {

                    const tid = `${pillar}-${idx}`;

                    return task.daily ? isDailyDoneToday(tid) : (taskStatus[tid] || false);

                }).length;

                const total = 8;

                const cells = [];

                

                for (let i = 0; i < total; i++) {

                    const isFilled = i < completed;

                    cells.push(`

                        <div class="aspect-square rounded-md ${isFilled ? pillarColors[pillar] : 'bg-slate-200 dark:bg-slate-700'}

                        </div>

                    </div>

                `;

            }).join('');

        }


        function updateStats() {

            let totalCompleted = 0;

            let totalTasks = 0;

            

            Object.keys(haradaData).forEach(pillar => {

                const tasks = haradaData[pillar];

                totalTasks += tasks.length;

                const completed = tasks.filter((task, idx) => {

                    const tid = `${pillar}-${idx}`;

                    return task.daily ? isDailyDoneToday(tid) : (taskStatus[tid] || false);

                }).length;

                totalCompleted += completed;

            });

            

            const overallPercentage = Math.round((totalCompleted / totalTasks) * 100);

            document.getElementById('overallProgress').textContent = overallPercentage + '%';

            

            Object.keys(haradaData).forEach(pillar => {

                updatePillarPercentage(pillar, haradaData[pillar].length);

            });

        }
